function renderMarkdown(raw) { if (!raw) return ''; // 1. جداسازی بلوک‌های کد برای جلوگیری از تغییر محتوای کدها var codeBlocks = []; var text = String(raw).replace(/ ```([\s\S]*?) ```/g, function(match, code) { var token = '___CODE_BLOCK_' + codeBlocks.length + '___'; codeBlocks.push('
' + escapeHtml(code.trim()) + '
'); return token; }); var inlineCodes = []; text = text.replace(/`([^`\n]+)`/g, function(match, code) { var token = '___INLINE_CODE_' + inlineCodes.length + '___'; inlineCodes.push('' + escapeHtml(code) + ''); return token; }); // 2. پاک‌سازی خطوط شارپ خالی یا ناقص (مانند ### . یا ####) text = text.replace(/^[ \t]*#{1,6}[ \t.*_–—\-]*$/gmi, ''); // 3. پاک‌سازی ستاره‌ها و علامت‌های ضخیم‌سازی خالی text = text.replace(/\*\*\s*\*\*/g, ''); text = text.replace(/^[ \t]*\*\*[ \t]*$/gmi, ''); // 4. اصلاح و پاک‌سازی فرمول‌های ناقص LaTeX // اصلاح fracهای فاقد بک‌اسلش در ابتدا text = text.replace(/(^|[^\\])frac\s*\{([^{}]+)\}\s*\{([^{}]+)\}/g, '$1\\frac{$2}{$3}'); // تبدیل کسرها به قالب خوانا و روان فارسی/ریاضی text = text.replace(/\\frac\s*\{([^{}]+)\}\s*\{([^{}]+)\}/g, '($1)/($2)'); // تبدیل علائم ضرب و توان LaTeX به نمادهای تمیز text = text.replace(/\\times/g, '×'); text = text.replace(/\\cdot/g, '·'); text = text.replace(/\\div/g, '÷'); // حذف دابل اسلش‌ها و پرانتزهای فرمول‌ساز \( و \) text = text.replace(/\\\(\s*/g, ''); text = text.replace(/\s*\\\)/g, ''); text = text.replace(/\\\[\s*/g, ''); text = text.replace(/\s*\\\]/g, ''); // 5. تبدیل فرمت‌های رایج متنی text = text.replace(/###\s+(.*)/g, '

$1

'); text = text.replace(/##\s+(.*)/g, '

$1

'); text = text.replace(/#\s+(.*)/g, '

$1

'); text = text.replace(/\*\*(.*?)\*\*/g, '$1'); text = text.replace(/\*(.*?)\*/g, '$1'); text = text.replace(/^[ \t]*[-*]\s+(.*)$/gmi, '
  • $1
    • '); // 6. مرتب‌سازی لیست‌ها و فواصل اضافه text = text.replace(/(
  • [\s\S]*?<\/li>)/g, '
      $1
    '); text = text.replace(/<\/ul>\s*
      /g, ''); text = text.replace(/\n{3,}/g, '\n\n'); text = text.replace(/\n/g, '
      '); // 7. بازگردانی بلوک‌های کد اولیه inlineCodes.forEach(function(codeHtml, index) { text = text.replace('___INLINE_CODE_' + index + '___', codeHtml); }); codeBlocks.forEach(function(blockHtml, index) { text = text.replace('___CODE_BLOCK_' + index + '___', blockHtml); }); return text.trim(); }
